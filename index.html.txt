<script>
let chart;
let historyData = [];
/* TAB */
function showTab(id,btn){
    document.querySelectorAll('.tab-content')
    .forEach(tab=>{
        tab.classList.remove('active');
    });
    document.querySelectorAll('.tab-btn')
    .forEach(button=>{
        button.classList.remove('active');
    });
    document.getElementById(id)
    .classList.add('active');
    btn.classList.add('active');
}
/* RISK CALCULATION */
function calculateRisk(){
    const name =
    document.getElementById('name').value || "Unknown";
    const weight =
    parseFloat(document.getElementById('weight').value);
    const angle =
    parseFloat(document.getElementById('angle').value);
    if(isNaN(weight) || isNaN(angle)){
        alert("Please enter valid values");
        return;
    }
    let risk =
    ((-0.000102 * angle * angle)
    + (0.00918 * angle)
    + 0.5)
    * (weight / 50);
    risk = parseFloat(risk.toFixed(4));
    const result =
    document.getElementById('result');
    let level = "";
    let className = "";
    let recommendation = "";
    if(risk < 0.5){
        level = "LOW RISK";
        className = "low";
        recommendation =
        "Maintain current posture and reposition every 2 hours.";
    }
    else if(risk < 0.8){
        level = "MODERATE RISK";
        className = "medium";
        recommendation =
        "Reduce pressure by adjusting the lying angle slightly.";
    }
    else{
        level = "HIGH RISK";
        className = "high";
        recommendation =
        "Immediate repositioning is recommended to reduce pressure.";
    }
    result.style.display = "block";
    result.className = `result ${className}`;
    result.innerHTML = `
        <h3>${level}</h3>
        <br>
        <p><b>Risk Score:</b> ${risk}</p>
        <p><b>Current Angle:</b> ${angle}°</p>
        <br>
        <p>${recommendation}</p>
    `;
    /* RECOMMENDED ANGLE */
    let recommendedAngle = 30;
    if(risk >= 0.8){
        recommendedAngle = 20;
    }
    else if(risk >= 0.5){
        recommendedAngle = 30;
    }
    else{
        recommendedAngle = angle;
    }
    document.getElementById('currentText')
    .innerHTML =
    `Current angle: ${angle}°`;
    document.getElementById('recommendedText')
    .innerHTML =
    `Recommended angle: ${recommendedAngle}°`;
    /* SAVE HISTORY */
    historyData.push({
        name:name,
        risk:risk,
        angle:angle,
        time:new Date().toLocaleString(),
        level:level
    });
    renderHistory();
    updateChart();
}
/* HISTORY */
function renderHistory(){
    const historyList =
    document.getElementById('historyList');
    historyList.innerHTML = "";
    historyData.forEach(item=>{
        let badgeClass = "";
        if(item.level.includes("LOW")){
            badgeClass = "low";
        }
        else if(item.level.includes("MODERATE")){
            badgeClass = "medium";
        }
        else{
            badgeClass = "high";
        }
        historyList.innerHTML += `
        <div class="history-item">
            <div>
                <strong>${item.name}</strong><br>
                <small>
                ${item.time} · ${item.angle}°
                </small>
            </div>
            <div class="badge ${badgeClass}">
                ${item.level} (${item.risk})
            </div>
        </div>
        `;
    });
}
/* CHART */
function updateChart(){
    const ctx =
    document.getElementById('riskChart');
    const labels =
    historyData.map((_,i)=>`Case ${i+1}`);
    const values =
    historyData.map(item=>item.risk);
    if(chart){
        chart.destroy();
    }
    chart = new Chart(ctx,{
        type:'line',
        data:{
            labels:labels,
            datasets:[{
                label:'Risk Score',
                data:values,
                borderWidth:3,
                fill:true,
                tension:0.4
            }]
        },
        options:{
            responsive:true
        }
    });
}
/* FEEDBACK STARS */
function rate(num){
    const stars =
    document.querySelectorAll('.star');
    stars.forEach((star,index)=>{
        if(index < num){
            star.classList.add('active');
        }
        else{
            star.classList.remove('active');
        }
    });
}
